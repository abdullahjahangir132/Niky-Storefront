ANIMATIONS — Must use Framer Motion correctly (no broken animations):
1. Page transitions: each page fades in with opacity 0→1 + y: 20→0, duration 0.4s

3. Navbar: mobile menu slides in from left (x: -100%→0)
4. Hero section: text appears word by word or line by line

6. Course cards: whileHover scale 1.02 + shadow increase

8. Modals: scale 0.9→1 + opacity 0→1
9. Sidebar: smooth collapse/expand with width animation
10. Toast notifications: slide in from top-right, auto-dismiss after 3 seconds

COMPLETE PRISMA SCHEMA:
model User {
  id        String   @id @default(cuid())
  name      String
  email     String   @unique
  password  String
  phone     String?
  role      Role     @default(USER)
  image     String?
  createdAt DateTime @default(now())
  enrollments Enrollment[]
  inquiries   Inquiry[]
}

enum Role { ADMIN USER VENDOR }

model Course {
  id          String   @id @default(cuid())
  title       String
  slug        String   @unique
  description String
  price       Float
  duration    String
  category    String
  thumbnail   String?
  teacher     String
  isActive    Boolean  @default(true)
  createdAt   DateTime @default(now())
  enrollments Enrollment[]
}

model Enrollment {
  id        String   @id @default(cuid())
  userId    String
  courseId  String
  user      User     @relation(fields:[userId], references:[id])
  course    Course   @relation(fields:[courseId], references:[id])
  paymentStatus String @default("PENDING")
  createdAt DateTime @default(now())
}

model Inquiry {
  id        String   @id @default(cuid())
  name      String
  email     String
  message   String
  status    String   @default("NEW")
  createdAt DateTime @default(now())
}

model InstituteInfo {
  id       String @id @default(cuid())
  phone    String @default("+92-300-0000000")
  email    String @default("info@aiacademy.com")
  address  String @default("123 Main Street, Karachi, Pakistan")
  hours    String @default("Mon-Sat: 9:00 AM - 7:00 PM")
  mapEmbed String @default("")
  updatedAt DateTime @updatedAt
}

model Vendor {
  id          String   @id @default(cuid())
  userId      String   @unique
  businessName String
  description String?
  isApproved  Boolean  @default(false)
  createdAt   DateTime @default(now())
}
